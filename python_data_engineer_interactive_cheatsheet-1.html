List — create, read, search, modify

listbasic
Ordered, mutable collection. Duplicates allowed.
Syntax
nums = [10, 20, 30]

# Access
nums[0]          # 10
nums[-1]         # 30
nums[1:3]        # [20, 30]

# Search
20 in nums       # True
nums.index(20)   # 1

# Add
nums.append(40)
nums.extend([50, 60])
nums.insert(1, 15)

# Modify
nums[0] = 99

# Remove
nums.remove(20)  # by value
nums.pop()       # last item
nums.pop(1)      # by index
del nums[0]

# Count / length
len(nums)
nums.count(40)

# Sort
nums.sort()
nums.sort(reverse=True)
sorted(nums)     # new list

# Reverse
nums.reverse()

# Copy
copy_nums = nums.copy()
DE pattern: use a list when order matters or when collecting rows/items sequentially.

Dictionary — key/value operations

dictlookup O(1) avg
Best for fast lookup, frequency counting, mapping, grouping, caching.
Syntax
d = {"a": 10, "b": 20}

# Read
d["a"]
d.get("c")            # None
d.get("c", 0)         # default

# Add / update

d["a"] = 99
d.update({"d": 40, "e": 50})

# Check
"a" in d

# Delete

d.pop("c", None)

# Keys / values / pairs
d.keys()
d.values()
d.items()

# Loop
for k, v in d.items():
    print(k, v)

# Frequency counter
counts = {}
for x in ["a","b","a"]:
    counts[x] = counts.get(x, 0) + 1

# Group values into lists
groups = {}
for dept, emp in [("IT","A"),("IT","B"),("HR","C")]:
    groups.setdefault(dept, []).append(emp)

For loops — common forms

forenumeratezip
Patterns
# Values
for x in [10,20,30]:
    print(x)

# Index + value
for i, x in enumerate([10,20,30]):
    print(i, x)

# Range
for i in range(5):        # 0..4
    pass

for i in range(1, 10, 2): # 1,3,5,7,9
    pass

# Two lists together
names = ["A","B"]
ages = [30,40]
for name, age in zip(names, ages):
    print(name, age)

# Dictionary
for k, v in {"a":1,"b":2}.items():
    print(k, v)

# break / continue
for x in range(10):
    if x == 3:
        continue
    if x == 7:
        break

LEFT ANTI / LEFT SEMI join

anti joinsemi join
Patterns
# LEFT ANTI: rows in left with NO match in right
anti = (
    left.merge(right[["id"]].drop_duplicates(),
               on="id", how="left", indicator=True)
        .query("_merge == 'left_only'")
        .drop(columns="_merge")
)

# Simpler anti for single key

# LEFT SEMI: rows in left WITH a match in right

LAG / LEAD

windowshift
Examples
df = df.sort_values(["id","date"])

# LAG(value, 1)
df["prev_value"] = df.groupby("id")["value"].shift(1)

# LEAD(value, 1)
df["next_value"] = df.groupby("id")["value"].shift(-1)

# Change from previous row
df["delta"] = df["value"] - df["prev_value"]

Running sum / cumulative aggregates

cumsumwindow
Examples

Rolling average / rolling window

rollingmoving average
Examples
df = df.sort_values(["id","date"])

# 3-row moving average per id
df["rolling_avg_3"] = (
    df.groupby("id")["value"]
      .transform(lambda s: s.rolling(3, min_periods=1).mean())
)

# 7-row rolling sum
df["rolling_sum_7"] = (
    df.groupby("id")["value"]
      .transform(lambda s: s.rolling(7, min_periods=1).sum())
)

Window aggregates with transform()

OVER PARTITION BYtransform
Examples
# AVG(salary) OVER(PARTITION BY dept)
df["dept_avg"] = (
    df.groupby("dept")["salary"]
      .transform("mean")
)

# MAX(salary) OVER(PARTITION BY dept)
df["dept_max"] = (
    df.groupby("dept")["salary"]
      .transform("max")
)

# Difference from group average
